# -------------------------------------------------------------------------
# Спецификация компонентов окружения
# -------------------------------------------------------------------------
# Данный набор библиотек является базовым индустриальным стандартом для
# разработки высокопроизводительных асинхронных веб-приложений. 
# Сочетание FastAPI и SQLAlchemy 2.0 обеспечивает полную изоляцию 
# бизнес-логики, валидацию входных данных и эффективную работу с СУБД 
# PostgreSQL в конкурентной среде без блокировки основного потока.


fastapi>=0.110.0
    # Современный, быстрый (высокопроизводительный) веб-фреймворк для 
    # создания API. Автоматически генерирует интерактивную документацию 
    # (Swagger UI / ReDoc) и обеспечивает декларативную обработку запросов.

uvicorn[standard]>=0.28.0
    # Высокопроизводительный ASGI-сервер, используемый для запуска приложения.
    # Опция [standard] включает оптимизированные бинарные зависимости 
    # (uvloop и httptools), обеспечивающие максимальную скорость обработки
    # сетевых сокетов.

pydantic>=2.6.0
    # Библиотека для валидации данных и управления конфигурацией на основе 
    # типизации Python. Версия 2.x полностью переписана на Rust, что 
    # гарантирует ультрабыструю сериализацию и десериализацию JSON-документов.

sqlalchemy[asyncio]>=2.0.28
    # Ключевая ORM-система для Python. Использование флага [asyncio] 
    # активирует полноценный асинхронный режим работы с базой данных, 
    # позволяя выполнять параллельные SQL-запросы в рамках архитектуры Master-Detail.

asyncpg>=0.29.0
    # Высокоэффективный, специализированный асинхронный драйвер для работы 
    # с PostgreSQL. Работает напрямую через бинарный протокол PostgreSQL, 
    # минуя классические клиентские библиотеки C, что делает его самым быстрым
    # драйвером на рынке.

anyio>=4.0.0
    # Унифицированный асинхронный слой ввода-вывода.
    # Используется FastAPI и SQLAlchemy как абстракция над asyncio/trio.
    # Позволяет безопасно управлять конкурентными задачами, таймаутами
    # и жизненным циклом фоновых операций без привязки к конкретному
    # асинхронному рантайму.
    # Явное объявление зависимости обязательно при прямом использовании
    # anyio в коде приложения.

python-multipart>=0.0.9
    # Системный парсер для обработки HTTP-запросов в формате multipart/form-data.
    # Необходим для корректного приема файлов и потоковых данных через API.

minio>=7.2.0
    # Официальный Python-клиент объектного хранилища MinIO,
    # совместимого с Amazon S3 API.
    # Обеспечивает потоковую загрузку и выгрузку бинарных объектов,
    # контроль целостности данных и работу с версиями объектов.
    # Используется для хранения исходных EML-файлов, тел писем,
    # вложений и результатов асинхронной обработки (OCR, парсинг).
